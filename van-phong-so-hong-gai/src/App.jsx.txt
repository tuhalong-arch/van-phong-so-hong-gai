import express from 'express';
import multer from 'multer';
import { GoogleGenAI } from '@google/genai';

const app = express();
const upload = multer({ dest: 'uploads/' });
const ai = new GoogleGenAI({ apiKey: process.env.GEMINI_API_KEY });

app.post('/api/parse-invitation', upload.single('file'), async (req, res) => {
  try {
    const file = req.file;
    
    // Gọi mô hình đa phương thức phân tích ảnh hoặc PDF của giấy mời
    const response = await ai.models.generateContent({
      model: 'gemini-2.5-flash',
      contents: [
        {
          role: 'user',
          parts: [
            {
              text: `Bạn là trợ lý hành chính văn phòng UBND phường. Hãy đọc văn bản giấy mời này và trích xuất đúng 4 trường thông tin theo định dạng JSON:
              - title: Nội dung cuộc họp
              - time: Thời gian họp cụ thể (giờ, ngày, tháng, năm)
              - location: Địa điểm tổ chức
              - attendees: Thành phần tham dự`
            },
            // Đính kèm buffer file đã tải lên
          ]
        }
      ],
      config: { responseMimeType: 'application/json' }
    });

    const parsedData = JSON.parse(response.text);
    res.json({ success: true, data: parsedData });
  } catch (error) {
    res.status(500).json({ success: false, message: error.message });
  }
});
